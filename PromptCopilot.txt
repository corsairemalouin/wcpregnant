package org.manu.service.wcpregnant.web;

import org.manu.service.wcpregnant.service.ToiletteService;
import org.manu.service.wcpregnant.web.dto.ToiletteDTO;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.web.bind.annotation.*;

import java.util.List;

@RestController
@RequestMapping("/api/toilettes")
public class ToiletteController {

    @Autowired
    private ToiletteService toiletteService;

    @GetMapping
    public List<ToiletteDTO> getAllToilettes() {
        return toiletteService.getAllToilettes();
    }

    @PostMapping
    public ToiletteDTO addToilette(@RequestBody ToiletteDTO toiletteDTO) {
        return toiletteService.addToilette(toiletteDTO);
    }

    @PutMapping("/{id}")
    public ToiletteDTO updateToilette(@PathVariable Long id, @RequestBody ToiletteDTO toiletteDTO) {
        return toiletteService.updateToilette(id, toiletteDTO);
    }

    @DeleteMapping("/{id}")
    public void deleteToilette(@PathVariable Long id) {
        toiletteService.deleteToilette(id);
    }
}

package org.manu.service.wcpregnant.service;

import org.manu.service.wcpregnant.bdd.dao.ToiletteDAO;
import org.manu.service.wcpregnant.bdd.repository.ToiletteRepository;
import org.manu.service.wcpregnant.service.bo.ToiletteBO;
import org.manu.service.wcpregnant.web.dto.ToiletteDTO;
import org.manu.service.wcpregnant.service.mapper.ToiletteMapper;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;

import java.util.List;
import java.util.stream.Collectors;

@Service
public class ToiletteService {

    @Autowired
    private ToiletteRepository toiletteRepository;

    @Autowired
    private ToiletteMapper toiletteMapper;

    public List<ToiletteDTO> getAllToilettes() {
        return toiletteRepository.findAll().stream()
                .map(toiletteMapper::toDTO)
                .collect(Collectors.toList());
    }

    public ToiletteDTO addToilette(ToiletteDTO toiletteDTO) {
        ToiletteDAO toiletteDAO = toiletteMapper.toDAO(toiletteDTO);
        ToiletteDAO savedToilette = toiletteRepository.save(toiletteDAO);
        return toiletteMapper.toDTO(savedToilette);
    }

    public ToiletteDTO updateToilette(Long id, ToiletteDTO toiletteDTO) {
        ToiletteDAO existingToilette = toiletteRepository.findById(id)
                .orElseThrow(() -> new ResourceNotFoundException("Toilette not found"));
        toiletteMapper.updateToiletteFromDTO(toiletteDTO, existingToilette);
        ToiletteDAO updatedToilette = toiletteRepository.save(existingToilette);
        return toiletteMapper.toDTO(updatedToilette);
    }

    public void deleteToilette(Long id) {
        toiletteRepository.deleteById(id);
    }
}
